# --- SEN2NAIPv2 access ---
# PINNED HARD, and the pin matters more than usual: we depend on
# `tacoreader.v1`, which upstream describes as legacy while it migrates to the
# v2 spec. The v2 API is incompatible -- its load() returns a TacoDataset with
# no .read() and does not accept the "tacofoundation:" protocol that resolves
# dataset names. If v1 is dropped in a future 2.x release, an unpinned install
# breaks the data layer with no warning. Verified working: 2.1.0.
tacoreader==2.1.0
# fsspec needs BOTH of these for the HTTP range reads tacoreader does. Without
# them tacoreader prints a warning, swallows the error, and fails later with a
# misleading "No objects to concatenate".
requests==2.32.3
aiohttp==3.11.11
# UNDECLARED DEPENDENCY OF tacoreader 2.1.0. `import tacoreader` fails outright
# with ModuleNotFoundError: No module named 'pydantic' on a clean install --
# tacoreader/dataset.py imports it but its package metadata does not list it.
# Verified by installing tacoreader alone into an empty venv. Do not remove.
pydantic==2.13.5

# --- Evaluation metrics ---
# scikit-image (above) provides SSIM. LPIPS needs its own package: it is the
# reference implementation, and its calibrated linear weights cannot be
# reconstructed from a torchvision backbone alone.
#
# IT DOWNLOADS PRETRAINED WEIGHTS ON FIRST USE (~233 MB of torchvision AlexNet,
# plus the small LPIPS linear layers that ship in the wheel), so it cannot run
# on a machine with no internet -- on Kaggle, enable the notebook's internet
# toggle, or set metrics.lpips.enabled=false. Everything else in
# src/metrics/ runs offline, and --smoke disables LPIPS for exactly this reason.
#
# lpips 0.1.4 declares torch/torchvision/numpy/scipy/tqdm, all already pinned
# above or installed alongside torch, so `pip install --no-deps lpips==0.1.4`
# is safe and avoids any chance of it resolving a different torch.
lpips==0.1.4

# --- Publishing the sample cache as a Kaggle Dataset ---
# Official Kaggle API client, driven by scripts/kaggle_upload.py -- you never
# type `kaggle` yourself. LOCAL ONLY: Kaggle notebooks already have it, and the
# script never runs there (the upload goes FROM this machine TO Kaggle).
#
# Needs an API token at ~/.kaggle/kaggle.json, from
# https://www.kaggle.com/settings/account -> API -> Create New Token.
# Every subcommand checks for it first and explains this if it is missing.

# VERIFIED 2.2.4: `datasets create` supports -p/--dir-mode/--keep-tabular and
# `datasets files --format csv --page-size` for the dry-run layout check. The
# script always invokes it as `python -m kaggle`, never the bare `kaggle`
# executable, which is NOT on PATH on this machine.
kaggle==2.2.4

# --- External benchmark: opensr-test ---------------------------------------
# The independent check on our own metrics (src/metrics/). ESA OpenSR, Aybar et
# al., IEEE JSTARS 2024. Used by src/eval/opensr_harness.py.
#
# INSTALL WITH --no-deps, AND PIN opencv/numpy WITH IT. This is not fussiness;
# MEASURED on this machine in that order:
#   1. `pip install opensr-test` wants open-clip-torch and openai-clip, which are
#      needed ONLY for the "clip" correctness distance that cfg.opensr_test does
#      not use. They are large and pull timm/ftfy.
#   2. Installing satalign pulled opencv-python 5.0.0.93, which REQUIRES
#      numpy>=2. pip duly upgraded numpy to 2.4.6 and broke scipy 1.14.1
#      (needs numpy<2.3) and the torch 2.5.1 build.
#   3. Downgrading numpy alone does not stick: opencv 4.14 also requires
#      numpy>=2, so pip re-upgrades it. Both must be pinned in ONE command.
#
#   .venv/Scripts/python.exe -m pip install opensr-test==1.3.3 --no-deps
#   .venv/Scripts/python.exe -m pip install satalign mpltern #       "opencv-python==4.10.0.84" "numpy==1.26.4"
#
# CONSEQUENCE, recorded in reports/day1_gate.md: the "clip" correctness
# distance is unavailable (VERIFIED: it raises ImportError for open_clip). The
# "lpips" distance still works, because lpips==0.1.4 is already pinned above for
# our own perceptual metric. "nd", the upstream default, is what we run.
opensr-test==1.3.3
satalign==0.1.17
mpltern==1.0.5
# Last opencv-python that accepts numpy 1.x. See the note above before bumping.
opencv-python==4.10.0.84
